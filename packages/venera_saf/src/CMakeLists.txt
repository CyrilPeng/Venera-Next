cmake_minimum_required(VERSION 3.16)
project(venera_saf VERSION 1.0.0 LANGUAGES CXX)
if(ANDROID)
  add_library(venera_saf SHARED handles.cpp native_bridge.cpp)
  target_link_options(venera_saf PRIVATE "-Wl,-z,max-page-size=16384")
else()
  add_library(venera_saf STATIC handles.cpp)
endif()
find_package(Threads REQUIRED)
target_link_libraries(venera_saf PRIVATE Threads::Threads)
target_compile_features(venera_saf PRIVATE cxx_std_17)
target_compile_definitions(venera_saf PRIVATE _FILE_OFFSET_BITS=64)
set_target_properties(venera_saf PROPERTIES CXX_VISIBILITY_PRESET hidden
  VISIBILITY_INLINES_HIDDEN YES POSITION_INDEPENDENT_CODE ON)
option(VENERA_SAF_BUILD_TESTS "Build descriptor ownership regressions" OFF)
if(VENERA_SAF_BUILD_TESTS)
  enable_testing()
  add_executable(saf_descriptor_test ../test/descriptor_test.cpp)
  target_link_libraries(saf_descriptor_test PRIVATE venera_saf)
  target_compile_features(saf_descriptor_test PRIVATE cxx_std_17)
  add_test(NAME saf_descriptor_ownership COMMAND saf_descriptor_test
    "${CMAKE_CURRENT_BINARY_DIR}/descriptor-test.bin")
endif()
